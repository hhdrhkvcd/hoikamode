    kill_country_leader = yes
    clr_country_flag = SER_tito_alive
    set_country_flag = SER_tito_dead
    create_country_leader = {
    name = "Aleksandar Vučić"
    desc = SER_LEADER_DESC
    picture = "gfx/leaders/BC/BC_SER_government.dds"
    expire = "2036.1.1.12"
    ideology = conservatism
    traits = { }
}
    set_politics = { ruling_party = democratic elections_allowed = no }
    set_popularities = { democratic = 55 fascism = 5 communism = 25 neutrality = 15 }
}

BC_initialize_CRO_army = {
    if = {
        limit = { NOT = { has_country_flag = BC_army_initialized } }
        set_country_flag = BC_army_initialized
        load_oob = "BC_CRO"
        add_equipment_to_stockpile = { type = infantry_equipment_1 amount = 1500 producer = CRO }
    }
}

BC_release_CRO = {
    if = {
        limit = { NOT = { country_exists = CRO } }
        release = CRO
    }
    if = { limit = { country_exists = CRO }
        CRO = {
            set_country_flag = BALKANS_INDEPENDENT
            BC_initialize_CRO_army = yes
        }
    }
    103 = { remove_core_of = SER }
    109 = { remove_core_of = SER }
    163 = { remove_core_of = SER }
    736 = { remove_core_of = SER }
    1306 = { remove_core_of = SER }
}

BC_dayton_settlement = {
    if = { limit = { country_exists = BOS country_exists = SRP }
        if = { limit = { SRP = { has_war_with = BOS } } SRP = { white_peace = BOS } }
        if = { limit = { 850 = { is_owned_by = SRP } } BOS = { transfer_state = 850 } }
        if = { limit = { SRP = { is_subject = no } } BOS = { puppet = SRP } }
        SRP = { set_politics = { ruling_party = neutrality elections_allowed = no } }
    }
}
