# Tito died on 4 May 1980. The federation survives for another decade.
country_event = {
    id = serbia_collapse.1
    title = serbia_collapse.1.t
    desc = serbia_collapse.1.d
    trigger = { tag = SER date > 1980.5.3 NOT = { has_country_flag = SER_tito_dead } }
    mean_time_to_happen = { days = 1 }
    fire_only_once = yes
    option = {
        name = serbia_collapse.1.a
        set_country_flag = SER_tito_dead
        clr_country_flag = SER_tito_alive
        add_stability = -0.08
        add_political_power = -50
        kill_country_leader = yes
        create_country_leader = {
            name = "Коллективное Президиум СФРЮ"
            desc = "SER_COLLECTIVE_DESC"
            picture = GFX_portrait_generic_communist
            expire = "1992.4.28.12"
            ideology = stalinism
            traits = { }
        }
        set_politics = { ruling_party = communism last_election = "1980.5.4" election_frequency = 12 elections_allowed = no }
        set_popularities = { democratic = 5 fascism = 2 communism = 82 neutrality = 11 }
    }
}
